  function emit(n,p,side,op){
    var out=[],cp=prec(n),wrap=(cp<p)||(side==="r"&&op==="-"&&cp===p);
    if(wrap){out.push(C("("));}
    if(n.t==="num"){
      var v=n.v;
      if(v<0){out.push(C("-"));v=-v;}
      var s=(Math.round(v)===v)?String(Math.round(v)):String(v);
      for(var i=0;i<s.length;i++){out.push(C(s.charAt(i)));}
    }else if(n.t==="var"){out.push(C("x"));}
    else if(n.t==="const"){out.push(C(n.name==="pi"?"π":n.name));}
    else if(n.t==="un"){out.push(C("-"));out=out.concat(emit(n.a,1.5,"l","+"));}
    else if(n.t==="bin"){
      var o=n.op;
      if(o==="+"||o==="-"){
        out=out.concat(emit(n.l,p,"l",o));out.push(C(o==="+"?"+":"-"));out=out.concat(emit(n.r,p,"r",o));
      }else if(o==="*"){
        var Aa=emit(n.l,2,"l","*"),Bb=emit(n.r,2,"r","*");
        out=out.concat(Aa);
        if(endD(Aa)&&startD(Bb)){out.push(C("*"));}
        out=out.concat(Bb);
      }else if(o==="/"){
        var f=mkFrac();f.n=emit(n.l,0,"l","/");f.d=emit(n.r,0,"r","/");out.push(f);
      }else if(o==="^"){
        out=out.concat(emit(n.l,3,"l","^"));
        var s2=mkSup(null);s2.e=emit(n.r,0,"r","^");out.push(s2);
      }
    }else if(n.t==="fn"){
      if(n.name==="sqrt"){var q=mkSqrt();q.b=emit(n.a,0,"l","/");out.push(q);}
      else if(n.name==="exp"){out.push(C("e"));var s3=mkSup(null);s3.e=emit(n.a,0,"r","^");out.push(s3);}
      else{out.push({t:"fn",n:n.name});out.push(C("("));out=out.concat(emit(n.a,0,"l","/"));out.push(C(")"));}
    }
    if(wrap){out.push(C(")"));}
    return out;
  }

  function makeGen(seed){
    var s=seed>>>0;
    function rnd(){s=(s*1664525+1013904223)>>>0;return s/4294967296;}
    function pick(a){return a[Math.floor(rnd()*a.length)];}
    var atoms=["x","(x+1)","(x+2)","(2x+1)","(x^2+1)","(x^2+2)","(x^2+x+1)","(x^3+1)"];
    var small=["x","(x+1)","(x-1)","(2x+1)","(x^2+1)","(x^2+2)"];
    var den=["(x^2+1)","(x^2+2)","(x+2)","(2x+3)","(x^2+x+1)"];
    var bounded=["sin(x)","cos(x)","atan(x)","tanh(x)","(x/2)","(x/3)","(x/4)"];
    var positive=["(x^2+1)","(x^2+2)","(x+2)","(2x+3)","abs(x)","(x^2+x+1)"];
    var trigArg=["x","(x/2)","(x/3)","atan(x)","sin(x)","cos(x)"];
    var lin=["x","(x/2)","(x/3)","(2x)","sin(x)","cos(x)","atan(x)"];
    function gen(d){
      if(d<=0){return pick(atoms);}
      var r=rnd();
      if(r<0.18){return pick(atoms);}
      if(r<0.40){return "("+gen(d-1)+pick(["+","-","*"])+gen(d-1)+")";}
      if(r<0.54){return "("+gen(d-1)+")/"+pick(den);}
      if(r<0.76){
        var f=pick(["sin","cos","tan","exp","sqrt","ln","atan","sinh","cosh","tanh"]);
        var arg;
        if(f==="exp"||f==="sinh"||f==="cosh"){arg=pick(bounded);}
        else if(f==="sqrt"||f==="ln"){arg=pick(positive);}
        else if(f==="tan"){arg=pick(trigArg);}
        else{arg=(rnd()<0.5)?pick(trigArg):gen(d-1);}
        return f+"("+arg+")";
      }
      if(r<0.90){return "("+pick(small)+")^"+pick(["2","3"]);}
      return pick(["2","3","e"])+"^("+pick(lin)+")";
    }
    return gen;
  }
  function alive(n,pts){
    var c=0;
    for(var i=0;i<pts.length;i++){
      var v=ev(n,pts[i]);
      if(isFinite(v)&&Math.abs(v)<1e7){c++;}
    }
    return c>=2;
  }
  function numericOk(ast,der,pts,tol){
    var ok=true,checked=0;
    for(var i=0;i<pts.length;i++){
      var x=pts[i],fv=ev(ast,x),u=ev(der,x),v=nd(ast,x);
      if(isFinite(fv)&&isFinite(u)&&isFinite(v)&&Math.abs(fv)<1e7&&Math.abs(u)<1e6&&Math.abs(v)<1e5){
        checked++;
        if(Math.abs(u-v)/Math.max(1,Math.abs(v))>tol){ok=false;}
      }
    }
    return ok&&checked>0;
  }
  function genCases(count,seed,order,pts){
    var gen=makeGen(seed),out=[],att=0,max=count*25;
    var SAFE=["x^3+2x","sin(x)cos(x)","e^x/(x^2+1)","ln(x^2+1)","sqrt(x^2+1)","x/(x^2+2)","tan(x/2)","atan(x^2)","sinh(x/2)","cosh(x/3)"];
    while(out.length<count&&att<max){
      att++;
      try{
        var src=gen(2+(att%2));
        var a=parse(src);
        if(!alive(a,pts)){continue;}
        var base=a,good=true;
        for(var k=0;k<order-1;k++){
          base=diff(base);
          if(!alive(base,pts)){good=false;break;}
        }
        if(!good){continue;}
        var sym=full(diff(base),{raz:true,fat:true});
        if(!numericOk(base,sym,pts,order>1?4e-3:3e-3)){continue;}
        out.push(src);
      }catch(e){}
    }
    var si=0;
    while(out.length<count){out.push(SAFE[si%SAFE.length]);si++;}
    return out;
  }
